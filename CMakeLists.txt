cmake_minimum_required(VERSION 3.20)
project(tcom LANGUAGES C)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "The CLI requires Windows and clang-cl.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(PNG REQUIRED)
find_path(WEBP_INCLUDE_DIR webp/encode.h REQUIRED)
find_library(WEBP_LIBRARY NAMES webp libwebp REQUIRED)
find_library(SHARPYUV_LIBRARY NAMES sharpyuv libsharpyuv REQUIRED)
add_executable(tcom tcom_dependency.c)
target_include_directories(tcom PRIVATE "${WEBP_INCLUDE_DIR}")
target_include_directories(tcom SYSTEM PRIVATE third_party)
target_compile_definitions(tcom PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(tcom PRIVATE PNG::PNG "${WEBP_LIBRARY}" "${SHARPYUV_LIBRARY}")
