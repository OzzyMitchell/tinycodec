cmake_minimum_required(VERSION 3.20)
project(tcom LANGUAGES C)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "The CLI requires Windows and clang-cl.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(PNG REQUIRED)
option(TCOM_READABLE "Build with the readable codec" OFF)
add_executable(tcom tcom_dependency.c)
if(TCOM_READABLE)
    target_compile_definitions(tcom PRIVATE TCOM_READABLE)
endif()
target_include_directories(tcom SYSTEM PRIVATE third_party)
target_compile_definitions(tcom PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(tcom PRIVATE PNG::PNG)
